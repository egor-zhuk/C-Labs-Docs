#### Упражнение 2. Добавление обработчика исключительных ситуаций
В программу добавлен код для деления двух целых чисел. Реализован блок `try-catch` для обработки ошибок (например, деления на ноль или ввода нечисловых данных).

**Код:**
```csharp
try
{
    Console.WriteLine("Please enter the first integer");
    string temp = Console.ReadLine();
    int i = int.Parse(temp);
    Console.WriteLine("Please enter the second integer");
    temp = Console.ReadLine();
    int j = int.Parse(temp);
    int k = i / j;
    Console.WriteLine("The result of dividing {0} by {1} is {2}", i, j, k);
}
catch (Exception e)
{
    Console.WriteLine("An exception was thrown: {0}", e);
}
```
