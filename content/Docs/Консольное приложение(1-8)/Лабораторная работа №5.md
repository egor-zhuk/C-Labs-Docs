#### Упражнение 1. Работа с массивами размерных типов (Чтение файла)
Реализована программа, принимающая имя файла через аргументы командной строки, считывающая его содержимое в массив символов и подсчитывающая статистику (гласные, согласные, строки).

**Код:**
```csharp
FileStream stream = new FileStream(fileName, FileMode.Open);
StreamReader reader = new StreamReader(stream);
size = (int)stream.Length;
contents = new char[size];
for (int i = 0; i < size; i++) contents[i] = (char)reader.Read();

static void Summarize(char[] contents)
{
	int vowels = 0, consonants = 0, lines = 0;
	foreach (char current in contents)
	{
		if (Char.IsLetter(current))
		{
			if ("AEIOUaeiou".IndexOf(current) != -1)
			{
				vowels++;
			}
			else
			{
				consonants++;
			}
		}
		else if (current == '\n')
		{
			lines++;
		}
	}
	Console.WriteLine("\n\nTotal no of characters: {0}", contents.Length);
	Console.WriteLine("Total no of vowels : {0}", vowels);
	Console.WriteLine("Total no of consonants: {0}", consonants);
	Console.WriteLine("Total no of lines : {0}", lines);
}
```

#### Упражнение 2. Перемножение матриц
Реализовано перемножение двух матриц 2x2. Матрицы вводятся пользователем.

**Код:**
```csharp
static int[,] Multiply(int[,] a, int[,] b)
{
    int[,] result = new int[2, 2];
    for (int r = 0; r < 2; r++)
    {
        for (int c = 0; c < 2; c++)
        {
            result[r, c] += a[r, 0] * b[0, c] + a[r, 1] *  b[1, c];
        }
    }
    return result;
}
```
